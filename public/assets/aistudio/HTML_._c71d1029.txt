let wakeLockSentinel = null, canvasStream = null, canvasInterval = null;

const activateWakeLock = async () => {
  if('wakeLock' in navigator){
    try { wakeLockSentinel = await navigator.wakeLock.request('screen'); }
    catch(e){ consoleLog('warn','WAKELOCK', 'Refusé : ' + e.message); }
  }
  if(!wakeLockSentinel){
    try {
      const canvas = $('noSleepCanvas');
      const video = $('noSleepVideo');
      if(!canvas || !video) return;
      const ctx = canvas.getContext('2d');
      if(!ctx) return;
      ctx.fillStyle = '#000'; ctx.fillRect(0,0,2,2);
      canvasStream = canvas.captureStream(1);
      let toggle = false;
      canvasInterval = setInterval(() => {
        toggle = !toggle;
        ctx.fillStyle = toggle ? '#000' : '#010101';
        ctx.fillRect(0,0,2,2);
      }, 1000);
      video.srcObject = canvasStream;
      video.muted = true;
      video.playsInline = true;
      await video.play().catch(() => {});
    } catch(e){ consoleLog('warn','WAKELOCK', 'Canvas fallback : ' + e.message); }
  }
};

const deactivateWakeLock = async () => {
  if(wakeLockSentinel){
    try { await wakeLockSentinel.release(); } catch(e){}
    wakeLockSentinel = null;
  }
  if(canvasInterval){ clearInterval(canvasInterval); canvasInterval = null; }
  const video = $('noSleepVideo');
  if(video){ try { video.pause(); video.srcObject = null; } catch(e){} }
  if(canvasStream){ try { canvasStream.getTracks().forEach(t => t.stop()); } catch(e){} canvasStream = null; }
};

  }

const getStoredKeys = () => {
  try {
    const raw = safeLocalStorage.get(API_KEYS_STORAGE);
    if(!raw) return [];
    const arr = JSON.parse(raw);
    return Array.isArray(arr) ? arr.filter(k => typeof k === 'string' && k.trim()) : [];
  } catch(e){ return []; }
};

const setStoredKeys = keys => safeLocalStorage.set(API_KEYS_STORAGE, JSON.stringify(keys));

const refreshKeyPool = () => {
  const stored = getStoredKeys();
  const old = state.keyPool || [];
  state.keyPool = stored.map(k => {
    const existing = old.find(p => p.key === k);
    return existing || { key:k, lastUsed:0, cooldownUntil:0, active:true, errorCount:0 };
  });
};

const hasActiveKey = () => state.keyPool.some(k => k.active);

const getReadyKey = () => {
  const now = Date.now();
  const active = state.keyPool.filter(k => k.active);
  if(active.length === 0) return { key:null, waitMs:0 };
  const ready = active.filter(k => !k.cooldownUntil || k.cooldownUntil <= now);
  if(ready.length === 0){
    const soonest = active.reduce((a,b) => (a.cooldownUntil||0) < (b.cooldownUntil||0) ? a : b);
    return { key:null, waitMs: Math.max(1000, (soonest.cooldownUntil||0) - now) };
  }
  ready.sort((a,b) => (a.lastUsed||0) - (b.lastUsed||0));
  return { key: ready[0], waitMs: 0 };
};

const acquireKey = async label => {
  while(true){
    if(state.stopRequested) throw new Error('Arrêt demandé');
    const { key, waitMs } = getReadyKey();
    if(key){ key.lastUsed = Date.now(); return key; }
    if(waitMs === 0) throw new Error('Aucune clé API active');
    consoleLog('warn','CLÉS','Toutes les clés en attente — ' + Math.round(waitMs/1000) + 's');
    setStatus('Patientez ' + Math.round(waitMs/1000) + 's…');
    await sleep(waitMs);
  }
};

const renderKeyList = () => {
  const el = $('key-list');
  if(!el) return;
  if(state.keyPool.length === 0){
    el.innerHTML = '<div style="font-size:.78rem;color:var(--dim);text-align:center;padding:1rem;">Aucune clé enregistrée.</div>';
    return;
  }
  const now = Date.now();
  el.innerHTML = state.keyPool.map((k,i) => {
    let stateHtml = '<span class="key-state">Active</span>';
    if(!k.active) stateHtml = '<span class="key-state dead">Inactive</span>';
    else if(k.cooldownUntil > now){
      const sec = Math.ceil((k.cooldownUntil - now)/1000);
      stateHtml = '<span class="key-state cool">Attente ' + sec + 's</span>';
    }
    return '<div class="key-item"><span class="key-id">#' + (i+1) + ' · ' + esc(k.key.slice(0,8)) + '…' + esc(k.key.slice(-4)) + '</span>' + stateHtml + '</div>';
  }).join('');
};

async function apiFetch(url, options, label, maxAttempts){
  options = options || {};
  label = label || 'API';
  maxAttempts = (typeof maxAttempts === 'number' && maxAttempts > 0) ? maxAttempts : 6;
  let lastErr = null;

  for(let attempt = 0; attempt < maxAttempts; attempt++){
    if(state.stopRequested) throw new Error('Arrêt demandé');
    const keyObj = await acquireKey(label);
    const method = options.method || 'GET';
    consoleLog('api','REQ ' + label, method + ' ' + url.slice(0, 60) + '… (essai ' + (attempt+1) + '/' + maxAttempts + ')');

    const headers = Object.assign({}, options.headers || {}, { 'Authorization': 'Bearer ' + keyObj.key });
    const finalOptions = Object.assign({}, options, { headers });

    try {
      const res = await fetch(url, finalOptions);
      let errorText = '';
      if(!res.ok){ try { errorText = (await res.clone().text()).slice(0, 400); } catch(e){} }
      consoleLog('api','RÉP ' + label, 'HTTP ' + res.status);

      if(res.ok){ keyObj.errorCount = 0; return res; }

      if([400, 401, 403, 404].includes(res.status)){
        if(res.status === 401 || res.status === 403){
          keyObj.errorCount++;
          if(keyObj.errorCount >= 3){
            keyObj.active = false;
            consoleLog('error','CLÉS','Clé désactivée après 3 erreurs');
            renderKeyList();
          }
        }
        const err = new Error('HTTP ' + res.status);
        err.status = res.status;
        throw err;
      }

      if(res.status === 429){
        const wait = Math.min(120, 20 + attempt * 15);
        keyObj.cooldownUntil = Date.now() + wait * 1000;
        consoleLog('warn','RATE','Attente ' + wait + 's');
        renderKeyList();
        continue;
      }

      if([500, 502, 503, 504, 520, 522, 524].includes(res.status)){
        const wait = Math.min(60, 8 + attempt * 6);
        consoleLog('warn','SERVICE','HTTP ' + res.status + ' — attente ' + wait + 's');
        setStatus('Service occupé — patientez…');
        await sleep(wait * 1000);
        continue;
      }

      await sleep(Math.min(60, 6 + attempt * 5) * 1000);
    } catch(e){
      if(e.message === 'Arrêt demandé') throw e;
      if(e.status && [400, 401, 403, 404].includes(e.status)) throw e;
      lastErr = e;
      const wait = Math.min(45, 4 + attempt * 4);
      consoleLog('warn','NETWORK', e.message + ' — retry ' + wait + 's');
      await sleep(wait * 1000);
    }
  }
  throw lastErr || new Error('Tentatives épuisées pour ' + label);
}

// JSON PARSER (5 stratégies)
// ══════════════════════════════════════════════════════════════════
function balanceJson(text){
  let ob=0, ok=0, inString=false, escape=false;
  for(let i=0;i<text.length;i++){
    const ch = text[i];
    if(escape){ escape=false; continue; }
    if(ch==='\\'){ escape=true; continue; }
    if(ch==='"'){ inString=!inString; continue; }
    if(inString) continue;
    if(ch==='{') ob++; else if(ch==='}') ob--;
    else if(ch==='[') ok++; else if(ch===']') ok--;
  }
  let r = text;
  if(inString) r += '"';
  while(ok>0){ r+=']'; ok--; }
  while(ob>0){ r+='}'; ob--; }
  return r;
}

function fixUnquotedKeys(text){
  let out='', i=0, inString=false, escape=false;
  const stack=[]; let lastStructural='{';
  while(i<text.length){
    const ch = text[i];
    if(escape){ out+=ch; escape=false; i++; continue; }
    if(ch==='\\'){ out+=ch; escape=true; i++; continue; }
    if(inString){ out+=ch; if(ch==='"') inString=false; i++; continue; }
    if(ch==='"'){ inString=true; out+=ch; i++; continue; }
    if(ch==='{'){ stack.push('{'); lastStructural='{'; out+=ch; i++; continue; }
    if(ch==='['){ stack.push('['); lastStructural='['; out+=ch; i++; continue; }
    if(ch==='}'){ stack.pop(); lastStructural='}'; out+=ch; i++; continue; }
    if(ch===']'){ stack.pop(); lastStructural=']'; out+=ch; i++; continue; }
    if(ch===','){ lastStructural=','; out+=ch; i++; continue; }
    if(ch===':'){ lastStructural=':'; out+=ch; i++; continue; }
    if(/\s/.test(ch)){ out+=ch; i++; continue; }
    const top = stack[stack.length-1];
    if((lastStructural==='{'||lastStructural===',') && top==='{' && /[A-Za-z_$]/.test(ch)){
      let j=i; while(j<text.length && /[A-Za-z0-9_$]/.test(text[j])) j++;
      let k=j; while(k<text.length && /\s/.test(text[k])) k++;
      if(text[k]===':'){ out += '"' + text.slice(i,j) + '"'; i = j; continue; }
    }
    out+=ch; i++;
  }
  return out;
}

function fixMissingCommas(text){
  let out='', i=0, inString=false, escape=false, justClosedValue=false;
  while(i<text.length){
    const ch = text[i];
    if(escape){ out+=ch; escape=false; i++; continue; }
    if(ch==='\\'){ out+=ch; escape=true; i++; continue; }
    if(inString){
      out+=ch;
      if(ch==='"'){
        inString=false;
        let j=i+1; while(j<text.length && /\s/.test(text[j])) j++;
        justClosedValue = (text[j] !== ':');
      }
      i++; continue;
    }
    if(/\s/.test(ch)){ out+=ch; i++; continue; }
    if(justClosedValue && ch!==',' && ch!=='}' && ch!==']' && ch!==':'){ out+=','; justClosedValue=false; }
    if(ch==='"'){ inString=true; justClosedValue=false; out+=ch; i++; continue; }
    if(ch==='{'){ justClosedValue=false; out+=ch; i++; continue; }
    if(ch==='['){ justClosedValue=false; out+=ch; i++; continue; }
    if(ch==='}'){ justClosedValue=true; out+=ch; i++; continue; }
    if(ch===']'){ justClosedValue=true; out+=ch; i++; continue; }
    if(ch===','){ justClosedValue=false; out+=ch; i++; continue; }
    if(ch===':'){ justClosedValue=false; out+=ch; i++; continue; }
    justClosedValue=false; out+=ch; i++;
  }
  return out;
}

function repairJson(text){
  if(!text) return null;
  let c = String(text).trim();
  const fm = c.match(/```(?:json)?\s*([\s\S]*?)```/);
  if(fm) c = fm[1].trim();
  const fb = c.indexOf('{');
  if(fb < 0) return null;
  c = c.slice(fb);
  const lb = c.lastIndexOf('}');
  if(lb > 0) c = c.slice(0, lb+1);
  let out='', inString=false, strDelim='"', escape=false;
  for(let i=0;i<c.length;i++){
    const ch = c[i];
    if(escape){
      if('"\\/bfnrtu'.includes(ch)){ out += '\\'+ch; } else { out += ch; }
      escape=false; continue;
    }
    if(ch==='\\'){ escape=true; continue; }
    if(ch==='"'||ch==="'"){
      if(!inString){ inString=true; strDelim=ch; out+='"'; continue; }
      if(ch===strDelim){ inString=false; out+='"'; continue; }
      if(ch==='"'){ out+='\\"'; continue; }
      out+="'"; continue;
    }
    if(inString){
      if(ch==='\n'){ out+='\\n'; continue; }
      if(ch==='\r'){ out+='\\r'; continue; }
      if(ch==='\t'){ out+='\\t'; continue; }
      const code = ch.charCodeAt(0);
      if(code < 32){ out += '\\u'+code.toString(16).padStart(4,'0'); continue; }
    }
    out += ch;
  }
  if(escape) out += '\\';
  if(inString) out += '"';
  try { out = fixUnquotedKeys(out); } catch(e){}
  try { out = fixMissingCommas(out); } catch(e){}
  out = out.replace(/,\s*([}\]])/g, '$1');
  return out;
}

function extractJson(text){
  if(!text) return null;
  const repaired = repairJson(text);
  if(!repaired) return null;
  try { return JSON.parse(repaired); }
  catch(e1){
    try { return JSON.parse(balanceJson(repaired)); }
    catch(e2){ consoleLog('error','PARSE','JSON irréparable'); return null; }
  }
}

function decodeCompositeId(composite){
  if(!composite || typeof composite !== 'string') return null;
  if(!composite.startsWith('video_')) return null;
  let b64 = composite.slice(6);
  while(b64.length % 4) b64 += '=';
  try {
    const decoded = atob(b64);
    const m = decoded.match(/video_id:([^;]+)/);
    if(m && m[1]) return m[1].trim();
  } catch(e){}
  return null;
}

function buildIdCandidates(videoIdObj){
  const ids = [];
  if(typeof videoIdObj === 'string'){
    ids.push({ label:'composite', value:videoIdObj });
    const inner = decodeCompositeId(videoIdObj);
    if(inner && inner !== videoIdObj) ids.push({ label:'inner', value:inner });
  } else if(videoIdObj && typeof videoIdObj === 'object'){
    if(videoIdObj.compositeId) ids.push({ label:'composite', value:videoIdObj.compositeId });
    if(videoIdObj.innerId) ids.push({ label:'inner', value:videoIdObj.innerId });
    const raw = videoIdObj.raw || {};
    for(const f of ['id','task_id','request_id']){
      if(raw[f] && !ids.some(x => x.value === raw[f])) ids.push({ label:'raw.'+f, value:raw[f] });
    }
  }
  const seen = new Set();
  return ids.filter(x => { if(seen.has(x.value)) return false; seen.add(x.value); return true; });
}

const POLL_URL_BUILDERS = [
  { name:'agnesapi', build:(id) => POLL_BASE + '?video_id=' + encodeURIComponent(id) + '&model_name=' + encodeURIComponent(MODEL_VIDEO) },
  { name:'v1-rest', build:(id) => API_BASE + '/videos/' + encodeURIComponent(id) },
  { name:'v1-status', build:(id) => API_BASE + '/videos/' + encodeURIComponent(id) + '/status' },
  { name:'v1-model', build:(id) => API_BASE + '/videos/' + encodeURIComponent(id) + '?model_name=' + encodeURIComponent(MODEL_VIDEO) }
];

async function pollVideo(videoIdObj, onProgress){
  const candidates = buildIdCandidates(videoIdObj);
  if(candidates.length === 0) throw new Error('Aucun ID exploitable');
  let rem = VIDEO_FIRST_POLL_DELAY_SEC;
  while(rem > 0){
    if(state.stopRequested) throw new Error('Arrêt demandé');
    onProgress('Préparation… (' + rem + 's)');
    await sleep(1000); rem--;
  }
  let working = { builder: state.workingPollUrlBuilder || null, id: null };
  for(let attempt = 0; attempt < VIDEO_MAX_POLL_ATTEMPTS; attempt++){
    if(state.stopRequested) throw new Error('Arrêt demandé');
    if(attempt > 0){
      const idx = Math.min(attempt - 1, VIDEO_POLL_INTERVALS.length - 1);
      let r = VIDEO_POLL_INTERVALS[idx];
      while(r > 0){
        if(state.stopRequested) throw new Error('Arrêt demandé');
        onProgress('Création… (' + r + 's)');
        await sleep(1000); r--;
      }
    }
    const approaches = [];
    if(working.builder && working.id) approaches.push({ builder: working.builder, id: working.id, priority: true });
    for(const builder of POLL_URL_BUILDERS){
      for(const cand of candidates){
        if(!approaches.some(a => a.builder === builder && a.id === cand.value)){
          approaches.push({ builder, id: cand.value, idLabel: cand.label });
        }
      }
    }
    for(const ap of approaches){
      if(state.stopRequested) throw new Error('Arrêt demandé');
      const url = ap.builder.build(ap.id);
      try {
        const res = await apiFetch(url, { method:'GET' }, 'Polling', 2);
        const d = await res.json();
        if(!working.builder || !working.id){
          working.builder = ap.builder;
          working.id = ap.id;
          state.workingPollUrlBuilder = ap.builder;
        }
        const status = d.status || 'unknown';
        const progress = d.progress || 0;
        onProgress('Création… ' + progress + '%');
        if(status === 'completed' || status === 'succeeded' || status === 'done'){
          const videoUrl = (d.metadata && d.metadata.url) || d.url || (d.output && d.output.url);
          if(!videoUrl) throw new Error("Terminé mais pas d'URL");
          incrementStats('videos');
          return videoUrl;
        }
        if(status === 'failed' || status === 'error' || status === 'cancelled'){
          throw new Error('Échec (' + status + ')');
        }
        break;
      } catch(e){
        if(e.message === 'Arrêt demandé') throw e;
        if(e.status === 404) continue;
        if(e.status === 401 || e.status === 403) continue;
        break;
      }
    }
  }
  throw new Error('Délai maximal dépassé');
}

    helmet: 'casque moto rétro chromé', outfit: 'kimono Shaolin rose poudré',
    physical_description: 'cute kawaii humanoid kitten with siamese cat face, wearing a chrome retro motorcycle helmet and a pastel pink Shaolin kimono, wes anderson deadpan'
  };
  const characters = [presenter];
  for(let i=0; i<mode.scenes-2; i++){
    characters.push({
      name: 'Reporter ' + (i+1),
      chat_face: chatBreeds[i % chatBreeds.length] + ' aux grands yeux',
      helmet: helmets[i % helmets.length],
      outfit: outfits[i % outfits.length],
      physical_description: 'cute kawaii humanoid kitten with ' + chatBreeds[i % chatBreeds.length] + ' cat face, wearing ' + helmets[i % helmets.length] + ' and ' + outfits[i % outfits.length] + ', wes anderson deadpan'
    });
  }
  const defaultShots = () => [
    { shot_number:1, camera:{ type:'wide symmetric', movement:'locked-off', framing:'centered wide' }, description:'wide symmetric tableau', duration_seconds:2 },
    { shot_number:2, camera:{ type:'medium', movement:'slow dolly-in', framing:'centered medium' }, description:'centered medium', duration_seconds:2 },
    { shot_number:3, camera:{ type:'close-up', movement:'locked-off', framing:'centered close-up' }, description:'close-up face', duration_seconds:2 },
    { shot_number:4, camera:{ type:'champ-contrechamp', movement:'locked-off', framing:'centered medium' }, description:'shot-reverse-shot', duration_seconds:2 },
    { shot_number:5, camera:{ type:'insert', movement:'locked-off', framing:'close-up object' }, description:'insert retro object', duration_seconds:2 }
  ];
  const scenes = [];
  for(let i=0; i<mode.scenes; i++){
    const isFirst = i === 0, isLast = i === mode.scenes - 1;
    const isPlateau = isFirst || isLast;
    const char = isPlateau ? presenter : characters[Math.min(i, characters.length-1)];
    const newsTitle = (newsItems && newsItems[i % newsItems.length]) ? newsItems[i % newsItems.length].title : '';
    let line;
    if(isFirst) line = 'Bonsoir, voici notre émission spéciale.';
    else if(isLast) line = 'Merci de nous avoir suivis, bonne soirée.';
    else if(newsTitle) line = newsTitle.length > 80 ? newsTitle.slice(0, 78) + '...' : newsTitle;
    else line = 'Voici notre reportage sur le terrain.';
    scenes.push({
      scene_number: i+1, type: isPlateau ? 'plateau' : 'reportage',
      character: char,
      decor: 'retro-futuristic pastel symmetric set, wes anderson style, palette ' + palette.label,
      action: 'the kitten character speaks to camera, deadpan',
      dialogue: { speaker: char.name, line_fr: line },
      text_overlay_fr: (subject || 'ÉMISSION').toUpperCase().slice(0, 40),
      shots: defaultShots()
    });
  }
  return {
    title: (subject || 'Ma création').slice(0, 60),
    logline: subject || 'Une émission cinématographique dans le style Wes Anderson.',
    characters, scenes
  };
}

function normalizeScript(script, mode, subject, palette, newsItems){
  if(!script || !script.scenes || !Array.isArray(script.scenes)) return null;
  if(!script.characters || !Array.isArray(script.characters) || script.characters.length === 0){
    script.characters = [{ name:'Présentateur', chat_face:'siamois', helmet:'casque rétro', outfit:'kimono Shaolin rose', physical_description:'kawaii humanoid kitten' }];
  }
  if(script.scenes.length > mode.scenes) script.scenes = script.scenes.slice(0, mode.scenes);
  while(script.scenes.length < mode.scenes){
    const idx = script.scenes.length;
    const srcScene = script.scenes[script.scenes.length - 1] || {};
    script.scenes.push({
      scene_number: idx+1, type: 'scene',
      character: srcScene.character || script.characters[0],
      decor: srcScene.decor || 'pastel symmetric set',
      action: 'the kitten speaks to camera',
      dialogue: { speaker: (srcScene.character&&srcScene.character.name)||'Présentateur', line_fr:'Suite de notre émission.' },
      text_overlay_fr: (subject||'ÉMISSION').toUpperCase().slice(0,40),
      shots: srcScene.shots || [{ shot_number:1, camera:{ type:'wide symmetric', movement:'locked-off', framing:'centered wide' }, description:'wide shot', duration_seconds:2 }]
    });
  }
  script.scenes.forEach((sc, i) => {
    sc.scene_number = i+1;
    if(!sc.character || !sc.character.name) sc.character = script.characters[Math.min(i, script.characters.length-1)];
    if(!sc.character.physical_description){
      const ch = sc.character;
      sc.character.physical_description = 'cute kawaii humanoid kitten with ' + (ch.chat_face||'siamois') + ' cat face, wearing ' + (ch.helmet||'casque rétro') + ' and ' + (ch.outfit||'kimono pastel') + ', wes anderson deadpan';
    }
    if(!Array.isArray(sc.shots) || sc.shots.length < 3){
      sc.shots = [
        { shot_number:1, camera:{ type:'wide symmetric', movement:'locked-off', framing:'centered wide' }, description:'wide symmetric', duration_seconds:2 },
        { shot_number:2, camera:{ type:'medium', movement:'slow dolly-in', framing:'centered medium' }, description:'medium shot', duration_seconds:2 },
        { shot_number:3, camera:{ type:'close-up', movement:'locked-off', framing:'centered close-up' }, description:'close-up', duration_seconds:2 }
      ];
    }
    if(!sc.dialogue || !sc.dialogue.line_fr){
      const newsTitle = (newsItems && newsItems[i % newsItems.length]) ? newsItems[i % newsItems.length].title : '';
      let fallbackLine;
      if(i === 0) fallbackLine = 'Bonsoir, voici notre émission.';
      else if(i === script.scenes.length-1) fallbackLine = 'Merci de nous avoir suivis.';
      else if(newsTitle) fallbackLine = newsTitle.length > 80 ? newsTitle.slice(0,78)+'...' : newsTitle;
      else fallbackLine = 'Voici notre reportage.';
      sc.dialogue = { speaker: sc.character.name || 'Présentateur', line_fr: fallbackLine };
    }
    if(!sc.decor) sc.decor = 'pastel retro-futuristic symmetric set';
    if(!sc.action) sc.action = 'the kitten speaks to camera, deadpan';
    if(!sc.text_overlay_fr) sc.text_overlay_fr = (subject || 'ÉMISSION').toUpperCase().slice(0, 40);
  });
  return script;
}
